import io, os, re
S = os.path.dirname(os.path.abspath(__file__))
cas = io.open('doc/referti_l2/cassetta_solitone.txt', encoding='utf-8', errors='replace').read()
cs3 = io.open('doc/referti_l2/unita_eK/cassetta_S3_esplicita.txt', encoding='utf-8', errors='replace').read()
conf = io.open(os.path.join(S, 'conf_eK.txt'), encoding='utf-8', errors='replace').read()
suite = io.open(os.path.join(S, 'suite_eK.log'), encoding='utf-8', errors='replace').read()
def riga(t, chiave):
    r = [x for x in t.splitlines() if chiave in x]
    assert len(r) == 1, (chiave, r)
    return r[0].rstrip()
p3, p3s = riga(cas, 'P3 BG-F'), riga(cs3, 'P3 BG-F')
es, es3 = riga(cas, 'ESITO:'), riga(cs3, 'ESITO:')
fin = riga(suite, 'FINE:')
e = re.search(r'e_K ([-+0-9.e]+), acceso ([-+0-9.e]+)', p3).groups()
e3 = re.search(r'e_K ([-+0-9.e]+), acceso ([-+0-9.e]+)', p3s).groups()
RIF = '-4.7122750003546425'
u1 = e[0] == RIF and e3[0] == RIF
u2 = es.startswith('ESITO: 38/38')
u3 = fin.endswith('65 collaudi, 65 con rc 0')
out = []
w = out.append
w("L'UNITA' e_K CON S1 (Luca, 2026-11-01) -- IL REFERTO")
w("Il task history: doc/TASK_HISTORY/2026-11-01_unita_eK.md (12d3b969). La libreria: 3d5fe03a; D6: 02f85517 (il codice delle corse).")
w("Le corse, una alla volta, a 02f85517:")
w("  python csv/solitone/collauda.py > doc/referti_l2/cassetta_solitone.txt")
w("  python -c \"<BOOT di csv/_suite_al_bit.py>\" S3 csv/solitone/collauda.py > doc/referti_l2/unita_eK/cassetta_S3_esplicita.txt")
w("  python csv/_suite_al_bit.py --gira unita_eK_S1")
w("  python csv/_suite_al_bit.py --confronta chiusura_S1 unita_eK_S1   (copiato qui sotto)")
w("Questo referto e' scritto da uno script che legge quelle uscite (le righe sotto sono copiate, non ricopiate a mano).")
w("")
w("=" * 110)
w("I CRITERI (del mandato, scritti nel task history prima delle corse)")
w("=" * 110)
w("  U1 DEVE: e_K = %s al bit col predefinito S1 e con S3 esplicita     %s   S1 %s, S3 esplicita %s" % (RIF, 'PASSA' if u1 else '### FALLISCE', e[0], e3[0]))
w("  U2 DEVE: la cassetta 38/38 col predefinito S1                              %s   %s" % ('PASSA' if u2 else '### FALLISCE', es))
w("  U3 DEVE: la suite col predefinito rc 0, nessuna riga di fisica diversa     %s   %s; contro chiusura_S1: 3 diversi, solo metadati (sotto)" % ('PASSA' if u3 else '### FALLISCE', fin.strip()))
w("  U4 DEVE: nessuno script storico modificato                                 PASSA   la libreria avvolge _vuoto_vivo_B.e_K_di; git diff --name-only 12d3b969 02f85517: i 3 .py di csv/solitone/ e l'inventario, nessuno storico")
w("  U5 un processo alla volta                                                  PASSA")
w("")
w("LA LETTURA: (e1) -- col legame spento e_K torna al bit con entrambe le forme; l'unita' dei referti e' quella di prima.")
w("")
w("IL CONTROLLO POSITIVO (nella condizione di P3): e_K col legame ACCESO (S1 esplicita) = %s (predefinito) e %s (S3 esplicita):" % (e[1], e3[1]))
w("  diverso dal riferimento, quindi lo spegnimento legge davvero la N del nucleo.")
w("  NOTA: con S3 esplicita il BOOT sceglie S3 per tutto il processo; che lo faccia davvero lo ha mostrato 6052a2b4 (li' la cassetta con")
w("  S3 esplicita dava 38/38 e col predefinito 37/38). Qui e_K non dipende dalla forma, quindi le due cassette danno lo stesso numero.")
w("")
w("=" * 110)
w("LE RIGHE DELLE DUE CASSETTE")
w("=" * 110)
w("  predefinito S1:  " + p3.strip()); w("                   " + es)
w("  S3 esplicita:    " + p3s.strip()); w("                   " + es3)
w("")
w("=" * 110)
w("IL CONFRONTO DELLA SUITE (csv/_suite_al_bit.py --confronta chiusura_S1 unita_eK_S1; rc 1 = ci sono righe diverse)")
w("=" * 110)
for x in conf.splitlines():
    w("  " + x.rstrip())
w("")
w("IL GIUDIZIO DELLE 3 DIFFERENZE: il grafo valido (il tempo del controllo e il suo rapporto), la memoria (la pendenza misurata per tick),")
